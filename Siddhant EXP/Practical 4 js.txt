    <script>
        function checkPalindrome() {
            const input = document.getElementById("userInput").value;
            // Clean the string: remove non-alphanumeric and lowercase
            const cleanStr = input.toLowerCase().replace(/[^a-z0-9]/g, '');
            // Reverse the string
            const reversedStr = cleanStr.split('').reverse().join('');
            const resultElement = document.getElementById("result");
            if (cleanStr !== '' && cleanStr === reversedStr) {
                resultElement.innerHTML = `- ${(input)} is a palindrome! (Reversed: ${reversedStr})`;
            } else if (cleanStr === "") {
                resultElement.innerHTML = "Please enter a value.";
            } else {
                resultElement.innerHTML = `- ${(input)} is not a palindrome. (Reversed: ${reversedStr})`;
            }
        }
    </script>
